cmake_minimum_required(VERSION 3.24)
project(TeraxShell LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Terax Explorer integration is Windows-only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(TeraxShellExtension SHARED ExplorerCommand.cpp exports.def)
target_compile_definitions(TeraxShellExtension PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(TeraxShellExtension PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /guard:cf)
target_link_options(TeraxShellExtension PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
target_link_libraries(TeraxShellExtension PRIVATE ole32 shell32 shlwapi uuid)
include(CTest)
if(BUILD_TESTING)
  add_executable(TeraxShellTests tests.cpp)
  target_compile_definitions(TeraxShellTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(TeraxShellTests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_link_libraries(TeraxShellTests PRIVATE ole32 shell32 shlwapi uuid)
  add_dependencies(TeraxShellTests TeraxShellExtension)
  add_test(NAME ExplorerCommand COMMAND TeraxShellTests)
endif()
